def extended_field(root,output):
    # Same observed trajectories, two explicit cost schedules. No decisions are retrained.
    frames={name:pd.read_csv(root/filename) for name,filename in author.FILES.items()}
    rowsets=[];tests=[];regressions=[];did=[]
    for label,h,b in [('author_h1_b9',1,9),('paper_h12_b88',12,88)]:
        data={}
        for group,frame in frames.items():
            m=author.field_metrics(frame).copy()
            m['holding_cost']*=h
            m['stockout_cost']*=b/9
            m['total_cost']=m.holding_cost+m.stockout_cost
            data[group]=m
            for metric in paper.METRICS:
                rowsets.append({'schedule':label,'h':h,'b':b,'group':group,'metric':metric,
                                'n':len(m),'mean':m[metric].mean()})
        for metric in paper.METRICS:
            t=data['treatment_post'][metric]
            c=data['control_post'][metric]
            test=stats.ttest_ind(t,c,equal_var=True)
            tests.append({'schedule':label,'metric':metric,'treatment_mean':t.mean(),'control_mean':c.mean(),
                'difference':t.mean()-c.mean(),'percent_change':100*(t.mean()-c.mean())/c.mean(),
                'pooled_t':test.statistic,'pooled_p':test.pvalue})
            post=pd.concat([data['treatment_post'].assign(e2e=1),data['control_post'].assign(e2e=0)],ignore_index=True)
            x=sm.add_constant(post[['e2e','ave_demand','vlt_num']],has_constant='add')
            for covariance in ['nonrobust','HC3']:
                fit=sm.OLS(post[metric],x).fit(cov_type=covariance)
                ci=fit.conf_int().loc['e2e']
                regressions.append({'schedule':label,'metric':metric,'intercept':True,'covariance':covariance,
                    'coefficient':fit.params['e2e'],'p':fit.pvalues['e2e'],'ci_low':ci.iloc[0],'ci_high':ci.iloc[1]})
            a=data['treatment_pre'][metric].mean();aa=t.mean()
            c0=data['control_pre'][metric].mean();cc=c.mean()
            did.append({'schedule':label,'metric':metric,'treatment_pre':a,'treatment_post':aa,
                'treatment_change':aa-a,'control_pre':c0,'control_post':cc,'control_change':cc-c0,
                'DiD':(aa-a)-(cc-c0),'audited_p':'not identified from supplied identity mapping'})
    for name,rows in [('field_schedules_means',rowsets),('table2_recomputed',tests),

def html_table(frame):
    def fmt(x):
        if isinstance(x,(float,np.floating)):
            if not np.isfinite(x):return '—'
            if 0<abs(x)<.0001:return f'{x:.3e}'
            return f'{x:,.4f}'.rstrip('0').rstrip('.')
        return str(x)
    return '<div class="table-wrap">'+frame.to_html(index=False,escape=True,border=0,formatters={c:fmt for c in frame.columns})+'</div>'

def inline_image(path,caption):

# %% Trace the original inventory code without changing its transitions
def trace_inventory(row, method):
    inv1, inv2, trace = [row['initial_stock'][0]], [], []
    rd = len(row['OPT_pred'])
    for r in range(rd):
        days = len(row['demand_hist'][r]) - (1 if r < rd-1 else 0)
        lead = int(round(row['vlt'][r]))
        for t in range(days):
            fallback = False
            if t == 0:
                if r == 0:
                    reference_stock = inv1[0]
                else:
                    try:
                        reference_stock = inv1[-lead-1]
                    except IndexError:
                        reference_stock = inv1[1]
                        fallback = True
                replen = int(round(row[method+'_pred'][r] - reference_stock))
            before = inv1[-1]
            demand = float(row['demand_hist'][r][t])
            after, counted, phase, cancelled_backlog = None, False, 'skipped_overlap', 0.
            if t < lead:
                if r == 0:
                    after = before-demand
                    inv1.append(after)
                    phase = 'before_first_arrival'
            elif t == lead:
                cancelled_backlog = max(-before, 0)
                after = (before if before >= 0 else 0) + replen - demand
                inv1.append(after)
                inv2.append(after)
                counted, phase = True, 'arrival'
            else:
                after = before-demand
                inv1.append(after)
                inv2.append(after)
                counted, phase = True, 'after_arrival'
            trace.append({'round':r, 'day_in_window':t, 'predicted_lead_days':lead,
                'prediction':float(row[method+'_pred'][r]), 'stock_reference':reference_stock,
                'computed_replenishment':replen, 'stock_before':before, 'demand':demand,
                'stock_after':after, 'phase':phase, 'counted_in_author_cost':counted,
                'backlog_removed_on_arrival':cancelled_backlog,
                'holding_cost':max(after,0) if counted else 0,
                'stockout_cost':9*max(-after,0) if counted else 0,
                'fallback_at_round_start':fallback if t == 0 else False})
    return [inv1[1:], inv2], pd.DataFrame(trace)

def field_metrics(frame, include_last=False, observed_denominator=False):

from functools import lru_cache
import json
import numpy as np

def interval_cost(demand, stock, order, h, b):
    end = stock + order - np.cumsum(demand)
    return float((h*np.maximum(end,0) + b*np.maximum(-end,0)).sum())

def closed_form(demand, stock, h, b, nonnegative=True):
    cumulative=np.cumsum(demand)
    k=int(b*len(demand)//(h+b))
    order=float(cumulative[k]-stock)
    return max(order,0.) if nonnegative else order

def greedy_solution(demand, cuts, stock, h, b):
    orders=[];cost=0.
    for left,right in zip(cuts[:-1],cuts[1:]):
        segment=demand[left:right]
        order=closed_form(segment,stock,h,b)
        orders.append(order)
        cost+=interval_cost(segment,stock,order,h,b)
        stock=stock+order-int(sum(segment))
    return cost,orders

def exhaustive_dp(demand, cuts, initial, h, b):
    """Enumerate all nonnegative integer actions up to a dominance bound."""
    @lru_cache(None)
    def dp(m, stock):
        if m==len(cuts)-1:
            return 0.
        left,right=cuts[m:m+2]
        segment=demand[left:right]
        # More than all remaining demand incurs avoidable holding cost, h>0.
        upper=max(sum(demand[left:])-stock,0)
        best=float('inf')
        for order in range(upper+1):
            cost=interval_cost(segment,stock,order,h,b)
            cost+=dp(m+1,stock+order-sum(segment))
            best=min(best,cost)
        return best
    return dp(0,initial)
